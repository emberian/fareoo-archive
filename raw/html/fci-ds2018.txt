class Cat s where
  type Arr s :: *
  dom :: (Arr s) ⟶ s
  cod :: (Arr s) ⟶ s
  idArr :: s ⟶ (Arr s)
  composeArr :: (Arr s) ⟶ (Arr s) ⤏ (Arr s)



Pure total functions: ⟶

Effectful functions: ⤏ (partial, non-det…)
